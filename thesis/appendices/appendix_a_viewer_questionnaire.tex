\chapter{Viewer Questionnaire}
\label{app:viewer_q}
\addcontentsline{toc}{chapter}{Appendix A: Viewer Questionnaire}

\noindent
This appendix reproduces the viewer questionnaire administered to
the 400-respondent sample in October 2024 -- February 2025.  The
instrument is reproduced in its English form; a Tamil-language
parallel version, prepared by back-translation, was used in
fieldwork.  The seven sections follow the structure described in
Section~\ref{sec:instruments} of Chapter~\ref{ch:methodology}.
Likert items are coded \textbf{1 = Strongly disagree} \dots
\textbf{5 = Strongly agree} unless otherwise noted.

\begin{enumerate}[itemsep=2pt]
  \item Age: \quad $\square$~18--24 \quad $\square$~25--34 \quad
        $\square$~35--44 \quad $\square$~45--54 \quad $\square$~55+
  \item Gender: \quad $\square$~Female \quad $\square$~Male \quad
        $\square$~Other / prefer not to say
  \item City / town of residence (Tamil Nadu / Puducherry):
        \rule{6cm}{0.4pt}
  \item Locality type: \quad $\square$~Metro \quad $\square$~Tier-2 city
        \quad $\square$~Tier-3 city \quad $\square$~Town \quad
        $\square$~Rural
  \item Highest educational qualification: \quad $\square$~School
        \quad $\square$~Undergraduate \quad $\square$~Postgraduate
        \quad $\square$~Professional \quad $\square$~Doctoral
  \item Monthly household income (INR): \quad $\square$~$<$25k
        \quad $\square$~25--50k \quad $\square$~50k--1L \quad

\end{enumerate}

\section*{Section B: Theatre Attendance}
\begin{enumerate}[resume,itemsep=2pt]
  \item Approximately how many Tamil films did you watch in a
        cinema hall in an average month \emph{in 2019}? \rule{2cm}{0.4pt}
  \item Approximately how many Tamil films did you watch in a
        cinema hall in an average month \emph{in 2024}? \rule{2cm}{0.4pt}
  \item Preferred theatre format: $\square$~Single-screen \quad
        $\square$~Multiplex \quad $\square$~Drive-in \quad
        $\square$~Premium-format (IMAX, Dolby, etc.)
  \item Typical ticket band you pay for a Tamil release (INR):
        $\square$~$<$100 \quad $\square$~100--200 \quad
        $\square$~200--400 \quad $\square$~$>$400
  \item Whom do you usually watch a Tamil film with at the theatre?
        $\square$~Alone \quad $\square$~Family \quad $\square$~Friends
        \quad $\square$~Partner
\end{enumerate}

\section*{Section C: OTT Subscription \& Consumption}
\begin{enumerate}[resume,itemsep=2pt]
  \item How many paid OTT subscriptions did your household hold
        \emph{in 2019}? \rule{2cm}{0.4pt}
  \item How many paid OTT subscriptions does your household hold
        \emph{today} (2024)? \rule{2cm}{0.4pt}
  \item Approximately how many hours per week did you watch OTT
        \emph{in 2019}? \rule{2cm}{0.4pt}
  \item Approximately how many hours per week do you watch OTT
        \emph{today} (2024)? \rule{2cm}{0.4pt}
  \item Tick all OTT platforms you use \emph{regularly} for
        Tamil-language films (multi-response):
        $\square$~Aha Tamil \quad $\square$~Sun NXT \quad
        $\square$~Disney+ Hotstar \quad $\square$~Amazon Prime Video \quad
        $\square$~Netflix \quad $\square$~ZEE5 \quad $\square$~JioCinema
        \quad $\square$~YouTube (free) \quad $\square$~Other

  \item Primary device on which you watch Tamil films \emph{today}:
        $\square$~Smartphone \quad $\square$~Smart TV \quad
        $\square$~Laptop / desktop \quad $\square$~Tablet \quad
        $\square$~STB / cable
\end{enumerate}

\section*{Section D: Genre, Language and Co-Viewing}
\begin{enumerate}[resume,itemsep=2pt]
  \item Allocate 100\% across the following genres for your
        \emph{2019} Tamil-film consumption: mass-action, family drama,
        romance, comedy, thriller / crime, investigative,
        realistic / indie, period / historical, sci-fi / fantasy,
        anthology.
  \item Repeat for your \emph{2024} consumption.
  \item Cross-language consumption \emph{2019}: tick all that apply
        \quad $\square$~Tamil only \quad $\square$~Tamil + Telugu
        \quad $\square$~Tamil + Malayalam \quad $\square$~Tamil + Hindi
        \quad $\square$~Tamil + Kannada \quad $\square$~Tamil +
        English / foreign.
  \item Repeat for \emph{2024}.
  \item Co-viewing context (most frequent, today): $\square$~Alone
        \quad $\square$~With family \quad $\square$~With friends
        \quad $\square$~With partner.
  \item Time of day for primary film viewing: $\square$~Morning
        \quad $\square$~Afternoon \quad $\square$~Evening \quad
        $\square$~Night (9 pm--12) \quad $\square$~Late night (12--3 am).
\end{enumerate}

\section*{Section E: Decision Drivers}
\begin{enumerate}[resume,itemsep=2pt]
  \item In 2019, the primary driver of my decision to watch a
        Tamil film was \emph{the lead star or director} (Y/N).
  \item In 2024, the primary driver of my decision to watch a
        Tamil film is \emph{the lead star or director} (Y/N).
  \item I trust word-of-mouth recommendations more now than in
        2019. (1--5)
  \item I trust OTT-platform recommendations more now than in
        2019. (1--5)
  \item I trust trade-press / critic reviews more now than in
        2019. (1--5)
  \item Social-media virality strongly influences my film choice.
        (1--5)
  \item The audio launch / music release strongly influences my
        film choice. (1--5)
  \item I am willing to wait for a film's OTT release rather than
        watch it in a theatre. (1--5)
\end{enumerate}

\section*{Section F: Willingness to Pay and Piracy}
\begin{enumerate}[resume,itemsep=2pt]
  \item Maximum I would pay for a TVOD rental of a new Tamil film:
        $\square$~$<$50 \quad $\square$~50--100 \quad $\square$~101--200
        \quad $\square$~201--300 \quad $\square$~$>$300.
  \item Frequency of piracy consumption \emph{in 2019}: $\square$~Never
        \quad $\square$~Rarely \quad $\square$~Occasionally \quad
        $\square$~Frequently.
  \item Repeat for \emph{2024}.
  \item I have churned (cancelled) at least one OTT subscription in
        the past 12 months. (Y/N)
\end{enumerate}

\section*{Section G: Theatre-to-OTT Window Preference}
\begin{enumerate}[resume,itemsep=2pt]
  \item Preferred theatre-to-OTT release window for a new Tamil
        film: $\square$~Same-day OTT \quad $\square$~1--2 weeks \quad
        $\square$~3--4 weeks \quad $\square$~5--8 weeks \quad
        $\square$~$>$8 weeks.
  \item Reasons (multi-response) for returning to theatres
        post-pandemic: big-screen experience; star release / FDFS;
        group / family outing; lack of OTT release; ticket
        affordability; cleanliness; convenience of OTT;
        content not engaging.
  \item Open comment: ``In 30 words or fewer, describe how your
        Tamil-film viewing has changed since 2019.''
\end{enumerate}
